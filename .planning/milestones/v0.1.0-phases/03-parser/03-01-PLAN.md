    **5. Update MakoLexer.flex** -- In the `<YYINITIAL>` state, replace the single generic rule:
    ```
    "<%" [a-zA-Z]  { yypushback(1); yybegin(TAG_ATTRS); return TAG_OPEN; }
    ```
    with 6 specific rules (order matters -- longest match first, JFlex uses longest-match so order among same-length doesn't matter, but keep them grouped):
    ```
    "<%def"            { yybegin(TAG_ATTRS); return TAG_OPEN_DEF; }
    "<%block"          { yybegin(TAG_ATTRS); return TAG_OPEN_BLOCK; }
    "<%inherit"        { yybegin(TAG_ATTRS); return TAG_OPEN_INHERIT; }
    "<%include"        { yybegin(TAG_ATTRS); return TAG_OPEN_INCLUDE; }
    "<%namespace"      { yybegin(TAG_ATTRS); return TAG_OPEN_NAMESPACE; }
    "<%page"           { yybegin(TAG_ATTRS); return TAG_OPEN_PAGE; }
    ```
    IMPORTANT: These rules must appear AFTER the `"<%doc>"`, `"<%!"` rules and BEFORE the `"<%" {WhiteSpace}` and `"<%" {LineTerminator}` anonymous code block rules. The old rule used `yypushback(1)` to push the first letter back so TAG_ATTRS could read it as TAG_ATTR_NAME. With per-tag rules, the tag keyword is consumed as part of the TAG_OPEN_xxx token itself, so there is NO yypushback. The TAG_ATTRS state will read the NEXT token (attribute name like `name`, or whitespace) directly.

    Note: The old lexer captured `<%` + one letter, pushed back the letter, and TAG_ATTRS read the full tag name as TAG_ATTR_NAME. Now TAG_OPEN_DEF captures `<%def` entirely, so the tag keyword is no longer a separate TAG_ATTR_NAME token. This is correct -- the grammar knows which tag it is from the token type.

    ```bnf
        TAG_OPEN_INCLUDE="TAG_OPEN_INCLUDE"
        TAG_OPEN_NAMESPACE="TAG_OPEN_NAMESPACE"
        TAG_OPEN_PAGE="TAG_OPEN_PAGE"

        CONTROL_LINE="CONTROL_LINE"
        CODE_OPEN="CODE_OPEN"
        CODE_CONTENT="CODE_CONTENT"
        MODULE_OPEN="MODULE_OPEN"
        MODULE_CONTENT="MODULE_CONTENT"
        CODE_CLOSE="CODE_CLOSE"
        LINE_COMMENT="LINE_COMMENT"
        DOC_OPEN="DOC_OPEN"
        DOC_CONTENT="DOC_CONTENT"
        DOC_CLOSE="DOC_CLOSE"
        TEMPLATE_TEXT="TEMPLATE_TEXT"
      ]
    }
    ```

    **Grammar rules:**
    ```bnf
    makoFile ::= item_*

    private item_ ::= def_tag
                     | block_tag
                     | inherit_tag
                     | include_tag
                     | namespace_tag
                     | page_tag
                     | expression
                     | control_line
                     | code_block
                     | module_block
                     | doc_comment
                     | line_comment_rule
                     | TEMPLATE_TEXT

    // ---- Named block tags (container tags with content + end tag) ----

    def_tag ::= TAG_OPEN_DEF tag_attribute* TAG_CLOSE item_* END_TAG
        {
          pin=1
          recoverWhile=tag_recover
          mixin="com.github.kimuth.jetbrainsmakotemplateplugin.lang.psi.impl.MakoDefTagMixin"
          implements="com.intellij.psi.PsiNamedElement"
          methods=[getName setName getNameIdentifier]
        }

    block_tag ::= TAG_OPEN_BLOCK tag_attribute* TAG_CLOSE item_* END_TAG
        {
          pin=1
          recoverWhile=tag_recover
          mixin="com.github.kimuth.jetbrainsmakotemplateplugin.lang.psi.impl.MakoBlockTagMixin"
          implements="com.intellij.psi.PsiNamedElement"
          methods=[getName setName getNameIdentifier]
        }

    // ---- Single-construct tags (no children, self-closing or simple close) ----

    inherit_tag ::= TAG_OPEN_INHERIT tag_attribute* TAG_CLOSE
        { pin=1  recoverWhile=tag_recover }

    include_tag ::= TAG_OPEN_INCLUDE tag_attribute* TAG_CLOSE
        { pin=1  recoverWhile=tag_recover }

    namespace_tag ::= TAG_OPEN_NAMESPACE tag_attribute* TAG_CLOSE
        { pin=1  recoverWhile=tag_recover }

    page_tag ::= TAG_OPEN_PAGE tag_attribute* TAG_CLOSE
        { pin=1  recoverWhile=tag_recover }

    // ---- Attribute helper (private = no PSI node) ----

    private tag_attribute ::= TAG_ATTR_NAME TAG_ATTR_EQ TAG_ATTR_VALUE
                            | TAG_ATTR_NAME

    // ---- Expression: ${...} ----

    expression ::= EXPR_START EXPR_CONTENT* (FILTER_SEP EXPR_CONTENT*)* EXPR_END
        { pin=1  recoverWhile=expression_recover }

    // ---- Control line: % keyword ... (single token) ----

    control_line ::= CONTROL_LINE

    // ---- Code block: <% ... %> ----

    code_block ::= CODE_OPEN CODE_CONTENT* CODE_CLOSE
        { pin=1  recoverWhile=tag_recover }

    // ---- Module block: <%! ... %> ----

    module_block ::= MODULE_OPEN MODULE_CONTENT* CODE_CLOSE
        { pin=1  recoverWhile=tag_recover }

    // ---- Doc comment: <%doc> ... </%doc> ----

    doc_comment ::= DOC_OPEN DOC_CONTENT* DOC_CLOSE
        { pin=1  recoverWhile=tag_recover }

    // ---- Line comment: ## ... (single token) ----
    // Named "line_comment_rule" to avoid conflict with the LINE_COMMENT token name

    line_comment_rule ::= LINE_COMMENT

    // ---- Recovery predicates ----
    // Negative predicates: recovery STOPS when any construct boundary token is seen

    private tag_recover ::= !(TAG_OPEN_DEF | TAG_OPEN_BLOCK | TAG_OPEN_INHERIT | TAG_OPEN_INCLUDE
                            | TAG_OPEN_NAMESPACE | TAG_OPEN_PAGE
                            | END_TAG | CONTROL_LINE | CODE_OPEN | MODULE_OPEN | DOC_OPEN
                            | EXPR_START | LINE_COMMENT)

    private expression_recover ::= !(EXPR_END | TAG_OPEN_DEF | TAG_OPEN_BLOCK | TAG_OPEN_INHERIT
                                   | TAG_OPEN_INCLUDE | TAG_OPEN_NAMESPACE | TAG_OPEN_PAGE
                                   | END_TAG | CONTROL_LINE | CODE_OPEN | MODULE_OPEN | DOC_OPEN
                                   | LINE_COMMENT)
    ```

    **Design notes in the BNF grammar:**
    - `generateTokens=false` prevents double-generating token constants that would cause silent parse failures (Pitfall 2 from research)
    - `pin=1` on every tag/expression rule commits the parser after matching the first token, enabling error recovery (PARS-05)
    - `recoverWhile` predicates include ALL construct-starting tokens to prevent over-consumption (Pitfall 3 from research)
    - `tag_attribute` is `private` so it doesn't create its own PSI node -- attributes are children of the tag node directly
    - `item_` is `private` (trailing underscore convention) so it doesn't create a wrapper PSI node
    - `line_comment_rule` avoids name clash with the `LINE_COMMENT` token
    - `def_tag` and `block_tag` have `mixin` + `implements` for PsiNamedElement (Plan 02 creates the mixin classes)
